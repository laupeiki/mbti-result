# -*- coding: utf-8 -*- """ MBTI 性格测试（命令行答题 + 自动生成 H5 结果页） 对接聚合数据「性格测试」接口： 1. 获取性格测试题目 https://apis.juhe.cn/fapig/character_test/questions 2. 提交答案获取结果 https://apis.juhe.cn/fapig/character_test/analysis 用法： python mbti_test.py 执行完成后，会在当前目录生成 H5 结果页面：mbti_result.html 用浏览器打开即可查看，也可部署到 GitHub Pages 转发给他人。 """ import json import requests # 基本参数配置 QUESTION_API = 'https://apis.juhe.cn/fapig/character_test/questions' # 获取题目接口 ANALYSIS_API = 'https://apis.juhe.cn/fapig/character_test/analysis' # 提交答案接口 API_KEY = '9a02fa8c2ab3473e7817adbfd1b60427' # 在个人中心->我的数据,接口名称上方查看 OUTPUT_HTML = 'mbti_result.html' # 生成的 H5 结果页文件名 def get_questions(api_key, level=''): """获取性格测试题目。level='' 返回4道题；level='senior' 返回12道题。""" request_params = { 'key': api_key, 'level': level, } response = requests.get(QUESTION_API, params=request_params) if response.status_code == 200: result = response.json() if result.get('error_code') == 0: return result['result'] raise RuntimeError('获取题目失败：错误码 {}，{}'.format( result.get('error_code'), result.get('reason'))) raise RuntimeError('请求异常，HTTP 状态码：{}'.format(response.status_code)) def get_analysis(api_key, answers): """提交答案，获取性格测试结果。answers 为 ia/ib 值列表。""" request_params = { 'key': api_key, 'answers': ','.join(answers), # 半角逗号分隔 } response = requests.get(ANALYSIS_API, params=request_params) if response.status_code == 200: result = response.json() if result.get('error_code') == 0: return result['result'] raise RuntimeError('提交失败：错误码 {}，{}'.format( result.get('error_code'), result.get('reason'))) raise RuntimeError('请求异常，HTTP 状态码：{}'.format(response.status_code)) def print_result(r): """在终端简要输出测试结果。""" print('\n' + '=' * 50) print(' 你的性格类型：{}（{}）'.format(r.get('alphabet', ''), r.get('occupation', ''))) print(' 英文人格：{}'.format(r.get('vocabulary', ''))) print('=' * 50) HTML_TEMPLATE = r""" 我的 MBTI 性格测试结果
MBTI PERSONALITY TEST
测试结果由聚合数据性格测试 API 生成
""" def build_html(r): """根据测试结果生成 H5 页面。""" data_json = json.dumps(r, ensure_ascii=False) return HTML_TEMPLATE.replace('__DATA__', data_json) def main(): print('=' * 50) print(' MBTI 性格测试（命令行版）') print('=' * 50) level = input('\n选择题目数量（回车=4道标准版，输入 senior=12道完整版）：').strip() # 1. 获取题目 print('\n正在获取题目...') try: questions = get_questions(API_KEY, level) except Exception as e: print(e) return print('共 {} 道题，请凭直觉作答。\n'.format(len(questions))) # 2. 逐题作答（选择A时填ia的值，选择B时填ib的值） answers = [] for i, q in enumerate(questions, 1): print('-' * 50) print('第 {}/{} 题：{}'.format(i, len(questions), q['q'])) print(' A. {}'.format(q['a'])) print(' B. {}'.format(q['b'])) while True: choice = input('请选择（A/B）：').strip().upper() if choice == 'A': answers.append(q['ia']) break elif choice == 'B': answers.append(q['ib']) break print('输入无效，请输入 A 或 B。') # 3. 提交答案，获取结果 print('\n正在分析你的性格...') try: result = get_analysis(API_KEY, answers) except Exception as e: print(e) return print_result(result) # 4. 生成 H5 结果页面 html = build_html(result) with open(OUTPUT_HTML, 'w', encoding='utf-8') as f: f.write(html) import os print('✅ H5 结果页面已生成：{}'.format(os.path.abspath(OUTPUT_HTML))) print(' 用浏览器打开即可查看，部署到 GitHub Pages 后可转发给他人。') print() print('#' * 60) print('# 以下为 index.html 完整内容，请全部复制（从 ） #') print('#' * 60) print(html) print('#' * 60) print('# 复制结束 #') print('#' * 60) if __name__ == '__main__': main()
